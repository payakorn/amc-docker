% \ChoiceColumns{n} -- typeset each question's CHOICES in n columns for the

% \ExamPart[cols]{title}{body} -- the body is the part's \ExamSection /
% \DrawGroup calls, written in config.tex.
\newcommand{\ExamPart}[3][2]{%
  \stepcounter{ExamPartNo}%
  \ChoiceColumns{1}%
  \ExamPartFormat{#2}%
  \AMC@mem@addsingle@ifneeded{\ExamPartFormatSheet{#2}}{\ExamPartTag}%
  \Exam@mem@part\Exam@mem@head
  \begin{center} \small (one correct answer per question, unless the question says ``select all that apply'') \end{center}
  \vspace{0.2cm}%
  \setlength{\columnsep}{.4cm}%
  \ifnum#1=1
    #3%
  \else
    \begin{multicols}{#1}%
      \raggedcolumns
      #3%
    \end{multicols}%
  \fi
}

% \SheetGrid{ncols}{filter} -- one block of the answer sheet, replaying just
% the items whose category matches. Used by \AnswerSheetLayout in config.tex.
\makeatletter
\newif\ifExam@firstgrid \Exam@firstgridtrue
\newcommand{\SheetGrid}[2]{%
  \ifExam@firstgrid \Exam@firstgridfalse \else \fi
  \setlength{\columnsep}{.5cm}%
  \ifnum#1=1
    \AMCformFilter{#2}%       % full width: digit grids, written answers
  \else
    \begin{multicols}{#1}%
      \AMCformFilter{#2}%
    \end{multicols}%
  \fi
}
% \SheetGridTitled[space]{ncols}{filter} -- like \SheetGrid, but the part
% headings go full width above the grid, and the grid starts on a new page
% unless [space] (default 8cm) is left. For narrow items such as number
% answers in columns of digits, set 4-6 across.
\newcommand{\SheetGridTitled}[3][8cm]{%
  \par\Needspace*{#1}%
  \AMCformFilter{(#3) && \InHeading}%
  \setlength{\columnsep}{.5cm}%
  \begin{multicols}{#2}%
    \AMCformFilter{(#3) && !\InHeading}%
  \end{multicols}%
}

% Framed answer-sheet boxes (name, student ID). AMC records every bubble and
% zone position with \pdfsavepos, which cannot see the coordinate shift
% tcolorbox's enhanced skin applies to its content, so an \AMCcode or
% \namefield placed inside the box is detected ~4 mm off in the scan. The
% content is therefore typeset into a saved box *outside* the tcolorbox, and
% the empty frame is drawn first, behind it.
% #1 title, #2 extra space below the content, #3 content.
\newsavebox{\SheetBoxContent}
\newcommand{\SheetBox}[3]{%
  \sbox{\SheetBoxContent}{%
    \begin{minipage}[t]{\dimexpr\linewidth-6mm-1pt\relax}#3\end{minipage}}%
  \par\noindent
  \begin{tcolorbox}[examtitled, title={#1},
      height=\dimexpr\ht\SheetBoxContent+\dp\SheetBoxContent+5mm+#2+1pt\relax]
  \end{tcolorbox}%
  \par\nointerlineskip
  \vspace{-\dimexpr\ht\SheetBoxContent+\dp\SheetBoxContent+2mm+#2+0.5pt\relax}%
  \noindent\hspace*{\dimexpr3mm+0.5pt\relax}\usebox{\SheetBoxContent}\par
  \vspace{\dimexpr2mm+#2\relax}}
