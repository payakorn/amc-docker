\element{ch02_stats}{
\begin{question}{STAT001}
Five students score 4, 6, 6, 7 and 9. What is the mode?
\begin{choices}
  \wrongchoice{4}
  \correctchoice{6}
  \wrongchoice{6.4}
  \wrongchoice{7}
\end{choices}
\end{question}
}

\element{ch02_stats}{
\begin{question}{STAT002}
A house-price list contains one mansion priced far above the rest. Which summary describes a typical price best?
\begin{choices}
  \wrongchoice{The mean}
  \correctchoice{The median}
  \wrongchoice{The maximum}
  \wrongchoice{The range}
\end{choices}
\end{question}
}

\element{ch02_stats}{
\begin{question}{STAT003}
A fair coin is tossed once. What is the probability of heads?
\begin{choices}
  \wrongchoice{0}
  \wrongchoice{0.25}
  \correctchoice{0.5}
  \wrongchoice{1}
\end{choices}
\end{question}
}

\element{ch02_stats}{
\begin{question}{STAT004}
Two shops have the same mean daily sales, but shop A's sales vary much more from day to day. Which statistic is larger for shop A?
\begin{choices}
  \wrongchoice{The mean}
  \wrongchoice{The median}
  \correctchoice{The standard deviation}
  \wrongchoice{The sample size}
\end{choices}
\end{question}
}

\element{ch02_stats}{
\begin{question}{STAT005}
Ice-cream sales rise on hotter days. What kind of relationship is this?
\begin{choices}
  \correctchoice{Positive correlation}
  \wrongchoice{Negative correlation}
  \wrongchoice{No correlation}
  \wrongchoice{A causal proof}
\end{choices}
\end{question}
}

\element{ch02_stats}{

  \wrongchoice{1/2}

\end{question}
