% Framed answer-sheet boxes (name, student ID). AMC records every bubble and
% zone position with \pdfsavepos, which cannot see the coordinate shift
% tcolorbox's enhanced skin applies to its content (it draws it inside a TikZ
% node), so an \AMCcode or \namefield placed inside the box is detected ~4 mm
% off in the scan. The content is therefore typeset into a saved box *outside*
% the tcolorbox, and the empty frame is drawn first, behind it.
% #1 title, #2 extra space below the content, #3 content.
\newsavebox{\SheetBoxContent}
\newcommand{\SheetBox}[3]{%
  \sbox{\SheetBoxContent}{%
    \begin{minipage}[t]{\dimexpr\linewidth-6mm-1pt\relax}#3\end{minipage}}%
  \par\noindent
  \begin{tcolorbox}[examtitled, title={#1},
      height=\dimexpr\ht\SheetBoxContent+\dp\SheetBoxContent+5mm+#2+1pt\relax]
  \end{tcolorbox}%
  \par\nointerlineskip
  \vspace{-\dimexpr\ht\SheetBoxContent+\dp\SheetBoxContent+2mm+#2+0.5pt\relax}%
  \noindent\hspace*{\dimexpr3mm+0.5pt\relax}\usebox{\SheetBoxContent}\par
  \vspace{\dimexpr2mm+#2\relax}}
